\newcommand{\multicancel@count}{1}
\newcommand{\multicancel@engine}{%
  \begin{tikzpicture}[baseline=(content.base), label distance=0pt]
    \node[inner sep=0pt, outer sep=0pt] (content) {\ensuremath{#1#2}};
    \foreach \i in {1,...,\multicancel@count} {
      \pgfmathsetmacro{\t}{\multicancel@count > 1 ? (\i - 1)/(\multicancel@count - 1) : 0.5}
      \pgfmathsetmacro{\offset}{(\t - 0.5) * 4pt} 
      \draw[line cap=round, shorten <= -1.5pt, shorten >= -1.5pt] 
      ($(content.south west) + (\offset, 0)$) -- ($(content.north east) + (\offset, 0)$);
    }
  \end{tikzpicture}%
}
\NewDocumentCommand{\multicancel}{O{1} m}{%
  \renewcommand{\multicancel@count}{#1}%
  \relax\ifmmode
    \mathpalette{\multicancel@engine}{#2}%
  \else
    \multicancel@engine{\textstyle}{\text{#2}}%
  \fi
}

\section{Introduction}
The \textsf{multicancel} package provides parallel multi-line cancellations across math expressions, large operators, and plain text. It prevents lines from slimming down dynamically when wrapping large content, maintains math styles, and functions across pdf\LaTeX, \XeLaTeX, and Lua\LaTeX.

\section{Usage}
\DescribeMacro{\multicancel}
\begin{verbatim}
\multicancel[n]{content}
\end{verbatim}

\section{Examples}
Default (1 line): $\multicancel{x+y}$\\
Double (2 lines): $\multicancel[2]{x+y}$\\
Quadruple (4 lines): $\multicancel[4]{x+y}$

\[
\multicancel[4]{\sum_{i=1}^{n} i} \quad \text{and} \quad \multicancel[3]{\int_{-\infty}^{\infty} e^{-x^2} \, dx}
\]
